      %\endnotemark[\ref{ {{- $.sourcesContainer.GetLabel $title -}} }]
    {{ end -}}
      {\color{source-link}\textsuperscript{\hyperref[ {{- $.sourcesContainer.GetLabelFor $title -}} ]{ {{- $key -}} }%
        }%
    }
  {{ end -}}
